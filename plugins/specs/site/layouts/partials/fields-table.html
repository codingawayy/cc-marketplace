{{/* Render fields array as a table with type linking. Parameters: - fields: Array of field objects with name, type, description - enums: Map of enum definitions (for type linking) - types: Map of type definitions (for type linking) */}} {{- $fields := .fields -}} {{- $enums := .enums | default dict -}} {{- $types := .types | default dict -}} {{ if $fields }}

Fields

{{- range $fields -}} {{- $fieldType := .type -}} {{- $typeDisplay := $fieldType -}} {{- $typeLink := "" -}} {{- $isOptional := strings.HasSuffix $fieldType "?" -}} {{- $cleanType := $fieldType | strings.TrimSuffix "?" -}} {{/* Parse enum format */}} {{- if strings.HasPrefix $cleanType "enum<" -}} {{- $enumName := $cleanType | strings.TrimPrefix "enum<" | strings.TrimSuffix ">" -}} {{- if index $enums $enumName -}} {{- $typeDisplay = $enumName | title -}} {{- if $isOptional -}} {{- $typeDisplay = printf "%s?" ($enumName | title) -}} {{- end -}} {{- $typeLink = "#enums" -}} {{- end -}} {{- end -}} {{/* Check if type references a custom type array like Link[] */}} {{- if and (not $typeLink) (strings.HasSuffix $cleanType "[]") -}} {{- $innerType := $cleanType | strings.TrimSuffix "[]" -}} {{- if index $types $innerType -}} {{- $typeLink = "#types" -}} {{- end -}} {{- end -}} {{- end -}}
Field Type Description
{{ .name }} {{- if $typeLink -}} {{ $typeDisplay }} {{- else -}} {{ $typeDisplay }} {{- end -}} {{ .description }}
{{ end }}
